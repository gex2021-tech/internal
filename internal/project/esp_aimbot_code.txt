static void draw_3d(aplayercontroller* controller,
                     uskeletalmeshcomponent* mesh,
                     fvector head_bone,
                     fvector base_bone,
                     flinearcolor color,
                     ucanvas* canvas)
{
    if (!controller || !mesh || !canvas) return;
    if (!head_bone.is_valid() || !base_bone.is_valid()) return;

    // 8 corner points of the 3D box
    fvector corners[8] = {
        {base_bone.x + 53, base_bone.y - 55, base_bone.z},
        {base_bone.x - 53, base_bone.y - 55, base_bone.z},
        {base_bone.x - 53, base_bone.y + 55, base_bone.z},
        {base_bone.x + 53, base_bone.y + 55, base_bone.z},
        {head_bone.x + 53, head_bone.y - 55, head_bone.z + 26},
        {head_bone.x - 53, head_bone.y - 55, head_bone.z + 26},
        {head_bone.x - 53, head_bone.y + 55, head_bone.z + 26},
        {head_bone.x + 53, head_bone.y + 55, head_bone.z + 26}
    };

    fvector2d screen[8];
    for (int i = 0; i < 8; i++) {
        if (!controller->project_world_location_to_screen(corners[i], screen[i], 0))
            return;
        if (!screen[i].is_valid()) return;
    }

    // Bottom face
    canvas->k2_drawline(screen[0], screen[1], 1.5f, color);
    canvas->k2_drawline(screen[1], screen[2], 1.5f, color);
    canvas->k2_drawline(screen[2], screen[3], 1.5f, color);
    canvas->k2_drawline(screen[3], screen[0], 1.5f, color);

    // Top face
    canvas->k2_drawline(screen[4], screen[5], 1.5f, color);
    canvas->k2_drawline(screen[5], screen[6], 1.5f, color);
    canvas->k2_drawline(screen[6], screen[7], 1.5f, color);
    canvas->k2_drawline(screen[7], screen[4], 1.5f, color);

    // Vertical edges
    canvas->k2_drawline(screen[0], screen[4], 1.5f, color);
    canvas->k2_drawline(screen[1], screen[5], 1.5f, color);
    canvas->k2_drawline(screen[2], screen[6], 1.5f, color);
    canvas->k2_drawline(screen[3], screen[7], 1.5f, color);
}

// Bone mapping
static fvector get_target_bone(uskeletalmeshcomponent* mesh, int bone) {
    switch (bone) {
    case 0: return mesh->get_bone_location(8);  // head
    case 1: return mesh->get_bone_location(6);  // neck
    case 2: return mesh->get_bone_location(4);  // chest
    default: return fvector();
    }
}

// Smoothing
static fvector smooth_aim(fvector target, fvector current, double smooth_factor) {
    fvector diff = target - current;
    // normalize angles
    float len = sqrtf(diff.x * diff.x + diff.y * diff.y + diff.z * diff.z);
    if (len > 0) { diff.x /= len; diff.y /= len; diff.z /= len; }
    return current + diff / smooth_factor;
}
